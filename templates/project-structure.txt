{{#if FRAMEWORK === "react"}}

│   ├── hooks/                 # useAestheticState 等
│   └── App.jsx
{{/if}}
{{#if FRAMEWORK === "vue"}}
├── src/
│   ├── components/            # *.vue 单文件组件
│   ├── composables/           # useAestheticState.ts
│   └── App.vue
{{/if}}
